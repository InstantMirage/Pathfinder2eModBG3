//REGION Flat Checks
// When any character starts casting a spell (target selected)
// give them a flat check value as a status. We will compare
// this value when we fire interrupts, and thus all the
// flat check conditions will automatically use the same roll
// and not need any stacking logic.

// Make flat check roll for caster
IF
UsingSpell((GUIDSTRING)_Caster, _, _, _, (INTEGER)_ID)
AND
Random(19, (INTEGER)_RollMinusOne)
AND
IntegerSum(_RollMinusOne, 1, (INTEGER)_Roll)
AND
IntegerProduct(_Roll, 6, _Duration)
AND
IntegerToReal(_Duration, (REAL)_DurationReal)
THEN
ApplyStatus(_Caster, "FLAT_CHECK_ROLL", _DurationReal, 1, _Caster);

//END_REGION

//REGION On change of obscurity, apply dim light status
IF
ObscuredStateChanged((GUIDSTRING)_Character, "HeavilyObscured")
THEN
ApplyStatus(_Character, "IN_DIM_LIGHT", -1.0);
RemoveStatus(_Character, "NOT_IN_DARKNESS");

IF
ObscuredStateChanged((GUIDSTRING)_Character, "LightlyObscured")
THEN
ApplyStatus(_Character, "NOT_IN_DARKNESS", -1.0);
RemoveStatus(_Character, "IN_DIM_LIGHT");

IF
ObscuredStateChanged((GUIDSTRING)_Character, "Clear")
THEN
ApplyStatus(_Character, "NOT_IN_DARKNESS", -1.0);
RemoveStatus(_Character, "IN_DIM_LIGHT");
//END_REGION

//REGION Concealment stacking

// Only out of combat, and only when no SG_Invisible marker is on: markers
// are handled by PF2e_Invisibility (in combat they get a drawn stand-in, not
// INVISIBLE). INVISIBLE grants this tag itself, so without the guard this

AND

//END_REGION

//REGION Base Game Sneaking
// Whenever a character enters combat while sneaking, remove the base
// game sneaking effect and immediately trigger a hide check.
IF
EnteredCombat((GUIDSTRING)_Character, _)
AND
HasActiveStatus(_Character, "SNEAKING", 1)
AND
IsPlayer((CHARACTER)_Character, 1)
THEN
RemoveStatus((GUIDSTRING)_Character, "SNEAKING");

// Repeat for NPCs
IF
EnteredCombat((GUIDSTRING)_Character, _)
AND
HasActiveStatus(_Character, "SNEAKING", 1)
AND
IsPlayer((CHARACTER)_Character, 0)
THEN
RemoveStatus((GUIDSTRING)_Character, "SNEAKING");

// when PF2e_Invisibility gives them INVISIBLE_COMBAT (its OnApplyFunctors

AND

THEN

// the hidden roll value.

AND
DB_HideCheckSchedule((GUIDSTRING)_Character, (GUIDSTRING)_Cause)
AND
CalculatePassiveSkill(_Cause, "Stealth", (INTEGER)_PassiveSkill)
AND

AND

AND

NOT DB_HideCheckSchedule(_Character, _Cause);
NOT DB_CheckConditionsForHideResult(_Character, _Cause);

NOT DB_HideCheckConcealed(_Character, _Cause);

// apply a status on the hider caused by the target we want to remove it

// I'm not entirely sure why, but directly applying the status breaks with
// non-hostile NPCs, resulting in enormous stacks of UNDETECTED that can't
// be removed. Applying it through a proxy like this fixes that. Sadly
// Osiris can't detect the cause of an existing status, so A and B work
// in tandem to make that condition check.
// Only while the hider is still Undetected (sneaking): if it attacked or
// stopped hiding before the helper landed, an UNDETECTED_FROM applied now
// would never be cleaned up.
IF
StatusApplied((GUIDSTRING)_Character, "UNDETECTED_FROM_HELPER_B", _, _)
AND
DB_UndetectedHelperSchedule(_Character, (GUIDSTRING)_Cause)
AND

NOT DB_UndetectedHelperSchedule(_Character, _Cause);

//REGION Undetected on back-to-back turns
// UNDETECTED_FROM makes the hider engine-invisible (UNDETECTED_ON_TURN) only
// on that viewer's turn. When two viewers act back to back, the next turn
// starts before the previous one ends, so the previous viewer's turn end
// strips the status the next viewer needs. Track whose turn it is and who
// the hider is Undetected to, and put it back.
IF
StatusApplied((GUIDSTRING)_Viewer, "UNDETECTED_FROM", (GUIDSTRING)_Hider, _)
THEN
DB_PF2_UndetectedFrom(_Viewer, _Hider);

// Lost Undetected during this viewer's own turn (a successful Seek): end the
// hider's per-turn invisibility now, so the viewer can act on it this turn.
// Must come before the rule below, which forgets the entries. StatusRemoved
// may not name the hider, so this ends it for all of this viewer's hiders.
IF
StatusRemoved((GUIDSTRING)_Viewer, "UNDETECTED_FROM", _, _)
AND
DB_PF2_CurrentTurn(_Viewer)
AND
DB_PF2_UndetectedFrom(_Viewer, (GUIDSTRING)_Hider)
THEN
NOT DB_PF2_UndetectedFrom(_Viewer, _Hider);
RemoveStatus(_Hider, "UNDETECTED_ON_TURN");

// StatusRemoved may not name the hider, so forget all of this viewer's
// entries; any hider still Undetected keeps its normal per-turn status.
IF
StatusRemoved((GUIDSTRING)_Viewer, "UNDETECTED_FROM", _, _)
AND
DB_PF2_UndetectedFrom(_Viewer, (GUIDSTRING)_Hider)
THEN
NOT DB_PF2_UndetectedFrom(_Viewer, _Hider);

IF
TurnStarted((GUIDSTRING)_Character)
AND
DB_PF2_CurrentTurn((GUIDSTRING)_Previous)
AND
_Previous != _Character
THEN
NOT DB_PF2_CurrentTurn(_Previous);

IF
TurnStarted((GUIDSTRING)_Character)
THEN
DB_PF2_CurrentTurn(_Character);

// Hide the hider as the viewer's turn starts, before its AI plans; the
// UNDETECTED_FROM tick lands later and the AI then re-plans mid-turn.
IF
TurnStarted((GUIDSTRING)_Viewer)
AND
DB_PF2_UndetectedFrom(_Viewer, (GUIDSTRING)_Hider)
THEN
ApplyStatus(_Hider, "UNDETECTED_ON_TURN", 6.0, 0, _Viewer);

IF
TurnEnded((GUIDSTRING)_Character)
AND
DB_PF2_CurrentTurn(_Character)
THEN
NOT DB_PF2_CurrentTurn(_Character);

// The causee is the viewer whose turn just ended; skip it so the current
// viewer's own turn end can't re-apply the status to itself.
IF
StatusRemoved((GUIDSTRING)_Hider, "UNDETECTED_ON_TURN", (GUIDSTRING)_Causee, _)
AND
DB_PF2_CurrentTurn((GUIDSTRING)_Viewer)
AND
_Viewer != _Causee
AND
DB_PF2_UndetectedFrom(_Viewer, _Hider)
THEN
ApplyStatus(_Hider, "UNDETECTED_ON_TURN", 6.0, 0, _Viewer);
//END_REGION

//REGION NPC Hide DC Calculation
// Originally we did these calcs in Khonsu, but the CPU load on NPC
// turns was untenably large. Instead, we calculate the hide DC once
// on the start of an NPC's turn, and just assume that that will be
// close enough throughout that turn. All the complexity here is just
// to ensure that the calculations for multiple entities with
// concurrent turns don't step on each other.
IF
TurnStarted((GUIDSTRING)_Character)
AND
IsPlayer((CHARACTER)_Character, 0)
THEN
ApplyStatus(_Character, "HIDE_NPC_CALC", 0.0);

IF
StatusApplied((GUIDSTRING)_Character, "HIDE_NPC_CALC", _, _)
AND
GUIDToString(_Character, (STRING)_GUIDString)
AND
Concatenate("HideCalculationGetDC", _GUIDString, (STRING)_HideDCString)
AND
Concatenate("HideCalculationFinal", _GUIDString, (STRING)_HideFinalString)
THEN
DB_HideCalculationNPC(_Character, 0, _HideDCString, _HideFinalString);
DB_CreateDiversionCalculationNPC(_Character, 0, _HideDCString, _HideFinalString);
IteratePlayerCharacters(_HideDCString, _HideFinalString);

StatusApplied((GUIDSTRING)_Character, "HIDE_CHECK_NPC_CONCEALED", (GUIDSTRING)_Cause, _)
THEN

EntityEvent((GUIDSTRING)_Character, (STRING)_HideDCString)
AND
DB_HideCalculationNPC((GUIDSTRING)_Cause, (INTEGER)_Current, _HideDCString, (STRING)_HideFinalString)
THEN
NOT DB_HideCalculationNPCConcealed(_Character, _Cause);

// Once every entity is checked, if no valid sources of DC were found, hiding
// is either impossible due to direct sight, or irrelevent. Set DC to 100.
IF
EntityEvent(_, _HideFinalString)
AND
DB_HideCalculationNPC(_Cause, (INTEGER)_Current, (STRING)_HideDCString, (STRING)_HideFinalString)
AND
_Current == 0
THEN
NOT DB_HideCalculationNPC(_Cause, _Current, _HideDCString, _HideFinalString);
DB_HideCalculationNPC(_Cause, 100, _HideDCString, _HideFinalString);

// For Create a Diversion, we just grab the perception DC.
IF
EntityEvent((GUIDSTRING)_Character, (STRING)_HideDCString)
AND
DB_CreateDiversionCalculationNPC((GUIDSTRING)_Cause, (INTEGER)_Current, _HideDCString, (STRING)_HideFinalString)
AND
CalculatePassiveSkill(_Character, "Perception", (INTEGER)_RawPerception)
AND
IntegerSum(_RawPerception, -2, (INTEGER)_PassivePerception)
AND
_PassivePerception > _Current
THEN
NOT DB_CreateDiversionCalculationNPC(_Cause, _Current, _HideDCString, _HideFinalString);
DB_CreateDiversionCalculationNPC(_Cause, _PassivePerception, _HideDCString, _HideFinalString);

// Set a status to store the DC for the character.
IF
EntityEvent(_, _HideFinalString)
AND
DB_HideCalculationNPC((GUIDSTRING)_Character, (INTEGER)_DC, (STRING)_HideDCString, _HideFinalString)
AND
DB_CreateDiversionCalculationNPC((GUIDSTRING)_Character, (INTEGER)_DiversionDC, (STRING)_HideDCString, _HideFinalString)
AND
IntegerProduct(_DC, 6, (INTEGER)_DCMult)
AND
IntegerToReal(_DCMult, (REAL)_DCReal)
AND
IntegerProduct(_DiversionDC, 6, (INTEGER)_DCDiversionMult)
AND
IntegerToReal(_DCDiversionMult, (REAL)_DCDiversionReal)
THEN
ApplyStatus(_Character, "HIDE_NPC_DC", _DCReal, 1, _Character);
ApplyStatus(_Character, "DIVERSION_NPC_DC", _DCDiversionReal, 1, _Character);
NOT DB_HideCalculationNPC(_Character, _DC, _HideDCString, _HideFinalString);
NOT DB_CreateDiversionCalculationNPC(_Character, _DiversionDC, _HideDCString, _HideFinalString);

//END_REGION

//REGION Flat Check Countering Conditions
// If we use Counterspell as our means of cancelling spells that fail
// their flat checks, we lose all functors of the spell, even those
// that consume a status on the user or deplete some other status-based
// resource. Instead, we want to manually block damage and certain
// conditions. Thankfully, SG_Condition exists to identify statuses that
// are broadly bad.

// Wipe DB of tracked statuses whenever a new spell is cast, specific to
// that combination of caster and target.
IF
UsingSpellOnTarget((GUIDSTRING)_Caster, (GUIDSTRING)_Target, _, _, _, _)
AND
DB_FlatCheckStatusesToClear(_Caster, _Target, (STRING)_Status)
THEN
NOT DB_FlatCheckStatusesToClear(_Caster, _Target, _Status);

// If a condition status is added to a character by a caster, add
// it to the DB if they have not already reacted.
IF
StatusApplied((GUIDSTRING)_Target, (STRING)_Status, (GUIDSTRING)_Caster, _)
AND
IsStatusFromGroup(_Status, "SG_Condition", 1)
AND
NOT DB_FlatCheckImmune(_Caster, _Target)
THEN
DB_FlatCheckStatusesToClear(_Caster, _Target, _Status);

// If the reaction has already fired (FLAT_CHECK_IMMUNITY), remove
// the status immediately.
IF
StatusApplied((GUIDSTRING)_Target, (STRING)_Status, (GUIDSTRING)_Caster, _)
AND
IsStatusFromGroup(_Status, "SG_Condition", 1)
AND
DB_FlatCheckImmune(_Caster, _Target)
THEN
RemoveStatus(_Target, _Status, _Target);

// Immunity is applied from the caster using a SWAP intermediate status.
// Add to a DB to track source.
IF
StatusApplied((GUIDSTRING)_Target, "FLAT_CHECK_IMMUNITY", (GUIDSTRING)_Caster, _)
THEN
DB_FlatCheckImmune(_Caster, _Target);

// When applied, remove any previously received conditions.
IF
StatusApplied((GUIDSTRING)_Target, "FLAT_CHECK_IMMUNITY", (GUIDSTRING)_Caster, _)
AND
DB_FlatCheckStatusesToClear(_Caster, _Target, (STRING)_Status)
THEN
RemoveStatus(_Target, _Status, _Target);
NOT DB_FlatCheckStatusesToClear(_Caster, _Target, _Status);

// The immunity status will expire, at which point we clear the DB.
IF
StatusRemoved((GUIDSTRING)_Target, "FLAT_CHECK_IMMUNITY", _, _)
AND
DB_FlatCheckImmune((GUIDSTRING)_Caster, _Target)
THEN
NOT DB_FlatCheckImmune(_Caster, _Target);

//END_REGION

//REGION Create a Diversion
// Reuses much of the same code as hiding, with some small changes.
IF
CastedSpell((GUIDSTRING)_Cause, "Target_CreateDiversion", _, _, _)
AND
TimerExists("DiversionCheckTimer", 0)
THEN
TimerLaunch("DiversionCheckTimer", 30);

// Create a distraction spawns an invisible helper that
// allows us to make humanoid enemies turn towards it. 
IF
EnteredLevel((GUIDSTRING)_Helper, CreateDiversionHelper_f08cb6ae-42b8-404f-a4c0-f57bd0347267, _)
THEN
DB_DiversionLocation(_Helper);

// Create a Diversion always checks against every enemy, and does
// not care about line of sight.
IF
TimerFinished("DiversionCheckTimer")
AND
DB_HideCheckSchedule((GUIDSTRING)_Character, (GUIDSTRING)_Cause)
AND
CalculatePassiveSkill(_Cause, "Deception", (INTEGER)_PassiveSkill)
AND
IntegerSubtract(_PassiveSkill, 10, (INTEGER)_ModifierSkill)
AND
GetStatusTurns(_Cause, "DIVERSION_CHECK_RESULT", (INTEGER)_Roll)
AND
IntegerSum(_Roll, _ModifierSkill, (INTEGER)_TotalRoll)
AND
DB_DiversionLocation((GUIDSTRING)_Helper)
THEN
PROC_ApplyHiddenDiversion(_Character, _Cause, _TotalRoll, _Roll);

// Cleanup and apply the stale status to debuff future attempts.
IF
TimerFinished("DiversionCheckTimer")
AND
DB_HideCheckSchedule((GUIDSTRING)_Character, (GUIDSTRING)_Cause)
AND
DB_DiversionLocation((GUIDSTRING)_Helper)
THEN
NOT DB_HideCheckSchedule(_Character, _Cause);
ApplyStatus(_Cause, "DIVERSION_STALE", 60.0, 1, _Character);
NOT DB_DiversionLocation(_Helper);

// A check of the roll against perception DC.
PROC
PROC_ApplyHiddenDiversion((GUIDSTRING)_Character, (GUIDSTRING)_Cause, (INTEGER)_Roll, (INTEGER)_Raw)
AND
HasActiveStatus(_Character, "DIVERSION_STALE", 0)
AND
CalculatePassiveSkill(_Character, "Perception", (INTEGER)_PassivePerception)
AND
_Roll >= _PassivePerception
THEN
ApplyStatus(_Character, "HIDDEN_FROM", -1.0, 0, _Cause);
PROC_TurnToDiversion(_Character);

// Crits always succeed
PROC
PROC_ApplyHiddenDiversion((GUIDSTRING)_Character, (GUIDSTRING)_Cause, (INTEGER)_Roll, (INTEGER)_Raw)
AND
_Raw == 20
THEN
ApplyStatus(_Character, "HIDDEN_FROM", -1.0, 0, _Cause);
PROC_TurnToDiversion(_Character);

// Characters that have recently used Create a Diversion are
// penalised. This should be a cirumcstance bonus to enemy
// perception DC, but we presently have nothing that would
// stack against that anyway.
PROC
PROC_ApplyHiddenDiversion((GUIDSTRING)_Character, (GUIDSTRING)_Cause, (INTEGER)_Roll, (INTEGER)_Raw)
AND
HasActiveStatus(_Cause, "DIVERSION_STALE", 1)
AND
CalculatePassiveSkill(_Character, "Perception", (INTEGER)_PassivePerception)
AND
IntegerSum(_PassivePerception, 4, (INTEGER)_PerceptionDC)
AND
_Roll >= _PerceptionDC
THEN
ApplyStatus(_Character, "HIDDEN_FROM", -1.0, 0, _Cause);
PROC_TurnToDiversion(_Character);

PROC
PROC_TurnToDiversion((GUIDSTRING)_Character)
AND
DB_DiversionLocation((GUIDSTRING)_Helper)
THEN
SteerTo((CHARACTER)_Character, _Helper, 0);
//END_REGION
