// at that distance. Apply a source status to it and await its
// summon event. Start a DB tracking the index.

DB_GustOfWindSpellsSummonSequence(_StoryID, _Owner, _Spell, (REAL)_XT, (REAL)_YT, (REAL)_ZT)

AND

AND
GetPosition(_Owner, (REAL)_XO, (REAL)_YO, (REAL)_ZO)
AND
QRY_VectorABByDistance(_XO, _YO, _ZO, _XT, _YT, _ZT, _Distance)
AND
DB_QRYRTN_VectorABByDistance((REAL)_XS, (REAL)_YS, (REAL)_ZS)
AND
CreateAt((GUIDSTRING)GustOfWind_Helper_74557a4c-6c00-44ea-8d42-30eec8010fa2, _XS, _YS, _ZS, 1, 0, "GustOfWindDummy", (GUIDSTRING)_Source)
THEN
NOT DB_GustOfWindCurrentSummon(_Distance);
DB_GustOfWindSpellsSource(_Source, _Owner);

ApplyStatus(_Source, "GUST_OF_WIND_SOURCE", 6.0, 0, _Owner);
RealtimeObjectTimerLaunch(_Source, "GustOfWindSummonStep", 100);

// Once the last helper is summoned, use the index value and
// add 1 to find the next distance in the DB. Do this until
// we reach the DB's limit. Each time, we also teleport the
// previous summon to its ideal co-ordinates, as the create
// at command puts it at the nearest empty space on the grid instead.

AND

//REGION Unwalkable wind logic
// When a character does not critically succeed against
// gust of wind, they cannot walk into the wind until
// the end of their turn. We have to do some nasty stuff
// to add and remove statuses that make walls walkable
// to achieve this.

// We have to track where each character received the
// block status from. THey can receive it from multiple
// sources, but we assume each caster only has 1 active
// wall.
IF
StatusApplied((GUIDSTRING)_Character, "GUST_OF_WIND_MOVE_BLOCK", (GUIDSTRING)_Source, (INTEGER)_StoryID)
THEN
DB_GustOfWindActiveBlockStatus(_Character, _Source, _StoryID);

IF
StatusRemoved((GUIDSTRING)_Character, "GUST_OF_WIND_MOVE_BLOCK", _, (INTEGER)_StoryID)
AND
DB_GustOfWindActiveBlockStatus(_Character, (GUIDSTRING)_Source, _StoryID)
THEN
NOT DB_GustOfWindActiveBlockStatus(_Character, _Source, _StoryID);

// Periodic timer to check active player.
IF
StatusApplied((GUIDSTRING)_Character, "GUST_OF_WIND_MOVE_BLOCK", (GUIDSTRING)_Source, (INTEGER)_StoryID)
AND
TimerExists("GustOfWindActivePlayerCheck", 0)
THEN
TimerLaunch("GustOfWindActivePlayerCheck", 500);
DebugText(_Character, "Timer Start");

IF
TimerFinished("GustOfWindActivePlayerCheck")
AND
DB_GustOfWindActiveBlockStatus((GUIDSTRING)_Character, (GUIDSTRING)_Source, (INTEGER)_StoryID)
THEN
TimerLaunch("GustOfWindActivePlayerCheck", 500);

// Whenever the active character changes, clear all walls, and
// trigger an update if the new character has the MOVE_BLOCK
// status and it is their turn.
IF
TimerFinished("GustOfWindActivePlayerCheck")
THEN
NOT DB_GustOfWindActiveCharacterFoundFlag(1);

IF
TimerFinished("GustOfWindActivePlayerCheck")
AND
DB_GustOfWindActiveBlockStatus((GUIDSTRING)_Character, (GUIDSTRING)_Source, (INTEGER)_StoryID)
AND
IsPlayer((CHARACTER)_Character, 1)
AND
GetReservedUserID(_Character, (INTEGER)_User)
AND
GetCurrentCharacter(_User, (CHARACTER)_CurrentCharacter)
AND
_Character == _CurrentCharacter
THEN
PROC_GustOfWindCurrentPlayerUpdate(_Character);
DB_GustOfWindActiveCharacterFoundFlag(1);

// If the active character is not found amongst those with
// the status, clear walls.
IF
TimerFinished("GustOfWindActivePlayerCheck")
AND
NOT DB_GustOfWindActiveCharacterFoundFlag(1)
THEN
PROC_GustOfWindClearWalls();

// If the active character is not found amongst those with
// the status, reset the active player tracker.
IF
TimerFinished("GustOfWindActivePlayerCheck")
AND
NOT DB_GustOfWindActiveCharacterFoundFlag(1)
AND
DB_GustOfWindOldPlayer((GUIDSTRING)_Character)
THEN
NOT DB_GustOfWindOldPlayer(_Character);

// Check if the character is the old active. If not, store.
PROC
PROC_GustOfWindCurrentPlayerUpdate((GUIDSTRING)_Character)
AND
NOT DB_GustOfWindOldPlayer(_Character)
THEN
DB_GustOfWindNewPlayer(_Character);

// If active character changed, remove old active.
PROC
PROC_GustOfWindCurrentPlayerUpdate((GUIDSTRING)_Character)
AND
DB_GustOfWindNewPlayer(_Character)
AND
DB_GustOfWindOldPlayer((GUIDSTRING)_Old)
THEN
NOT DB_GustOfWindOldPlayer(_Old);

// Do updates if the character changed. Note that the force update
// status only makes an update if it is the character's turn.
PROC
PROC_GustOfWindCurrentPlayerUpdate((GUIDSTRING)_Character)
AND
DB_GustOfWindNewPlayer(_Character)
THEN
PROC_GustOfWindClearWalls();
ApplyStatus(_Character, "GUST_OF_WIND_FORCE_UPDATE", 0.0);

// Remove the new character form storage ready for next change.
PROC
PROC_GustOfWindCurrentPlayerUpdate((GUIDSTRING)_Character)
AND
DB_GustOfWindNewPlayer(_Character)
THEN
DB_GustOfWindOldPlayer(_Character);
NOT DB_GustOfWindNewPlayer(_Character);

// On any turn start, clear out all walls. A passive will trigger
// further updates.
IF
TurnStarted(_)
THEN
PROC_GustOfWindClearWalls();

// The status is only applied if it is the character's
// turn in combat. We only need to check that PCs are
// actively being controlled. NPCs just pass through.
// We trigger on status loss so that this always occurs
// after the turn start clear.
IF
StatusRemoved((GUIDSTRING)_Character, "GUST_OF_WIND_WALL_UPDATE", _, _)
AND
IsPlayer((CHARACTER)_Character, 0)
AND
DB_GustOfWindActiveBlockStatus((GUIDSTRING)_Character, (GUIDSTRING)_Source, _)
THEN
PROC_GustOfWindUpdateWalls(_Character, _Source);

// A PC gains the status while they are the active
// character triggers an update.
IF
StatusRemoved((GUIDSTRING)_Character, "GUST_OF_WIND_WALL_UPDATE", _, _)
AND
IsPlayer((CHARACTER)_Character, 1)
AND
GetReservedUserID(_Character, (INTEGER)_User)
AND
GetCurrentCharacter(_User, (CHARACTER)_CurrentCharacter)
AND
_Character == _CurrentCharacter
AND
DB_GustOfWindActiveBlockStatus(_Character, (GUIDSTRING)_Source, _)
THEN
PROC_GustOfWindUpdateWalls((GUIDSTRING)_Character, _Source);

// Whenever an update is triggered...
// Clear the current walls DB, for the relevent source.
PROC
PROC_GustOfWindUpdateWalls((GUIDSTRING)_Character, (GUIDSTRING)_Source)
AND
DB_GustOfWindCurrentSolidWalls(_Source, (GUIDSTRING)_Summon)
THEN
NOT DB_GustOfWindCurrentSolidWalls(_Source, _Summon);

// Identify which walls are closer to the source than
// the triggering character. We essentially work out
// how far away the character is in the axis of the
// spell, and subtract how far off-axis they are. Any
// wall that is closer than that to the source (which
// is 10m behind the original cast) becomes unwalkable.
PROC
PROC_GustOfWindUpdateWalls((GUIDSTRING)_Character, (GUIDSTRING)_Source)
AND
GetPosition(_Character, (REAL)_XC, (REAL)_YC, (REAL)_ZC)
AND
DB_GustOfWindSpellsSummons((GUIDSTRING)_Summon, _Source, (GUIDSTRING)_SpellSource, (REAL)_Distance)
AND
GetPosition(_SpellSource, (REAL)_XS, (REAL)_YS, (REAL)_ZS)
AND
RealSubtract(_XC, _XS, (REAL)_XA)
AND
RealSubtract(_YC, _YS, (REAL)_YA)
AND
RealSubtract(_ZC, _ZS, (REAL)_ZA)
AND
GetPosition(_Summon, (REAL)_XSu, (REAL)_YSu, (REAL)_ZSu)
AND
RealSubtract(_XSu, _XS, (REAL)_XB)
AND
RealSubtract(_YSu, _YS, (REAL)_YB)
AND
RealSubtract(_ZSu, _ZS, (REAL)_ZB)
AND
QRY_VectorProjection(_XA, _YA, _ZA, _XB, _YB, _ZB)
AND
DB_QRYRTN_VectorProjection((REAL)_Projection)
AND
QRY_VectorRejection(_XA, _YA, _ZA, _XB, _YB, _ZB)
AND
DB_QRYRTN_VectorRejection((REAL)_Rejection)
AND
RealSubtract(_Projection, _Rejection, (REAL)_DistanceScore)
AND
// Add an extra 1 here so that walls we are standing in are walkable.
RealSum(_Distance, 11.0, (REAL)_SummonDistance)
AND
_DistanceScore > _SummonDistance
THEN
DB_GustOfWindCurrentSolidWalls(_Source, _Summon);
DebugText(_Summon, "Closer");

// Add the unwalkable status top any new walls.
PROC
PROC_GustOfWindUpdateWalls((GUIDSTRING)_Character, (GUIDSTRING)_Source)
AND
DB_GustOfWindCurrentSolidWalls(_Source, (GUIDSTRING)_Summon)
AND
NOT DB_GustOfWindOldSolidWalls(_Source, _Summon)
THEN
ApplyStatus(_Summon, "GUST_OF_WIND_UNWALKABLE", -1.0);

// Remove the unwalkable status from any old walls.
PROC
PROC_GustOfWindUpdateWalls((GUIDSTRING)_Character, (GUIDSTRING)_Source)
AND
DB_GustOfWindOldSolidWalls(_Source, (GUIDSTRING)_Summon)
AND
NOT DB_GustOfWindCurrentSolidWalls(_Source, _Summon)
THEN
RemoveStatus(_Summon, "GUST_OF_WIND_UNWALKABLE");

// After, clear out the old walls DB.
PROC
PROC_GustOfWindUpdateWalls((GUIDSTRING)_Character, (GUIDSTRING)_Source)
AND
DB_GustOfWindOldSolidWalls(_Source, (GUIDSTRING)_Summon)
THEN
NOT DB_GustOfWindOldSolidWalls(_Source, _Summon);

// Repopulate the old walls DB with the contents of the
// new walls DB, to use next time.
PROC
PROC_GustOfWindUpdateWalls((GUIDSTRING)_Character, (GUIDSTRING)_Source)
AND
DB_GustOfWindCurrentSolidWalls(_Source, (GUIDSTRING)_Summon)
THEN
DB_GustOfWindOldSolidWalls(_Source, _Summon);

// Clear out all walls
PROC
PROC_GustOfWindClearWalls()
AND
DB_GustOfWindCurrentSolidWalls((GUIDSTRING)_Source, (GUIDSTRING)_Summon)
THEN
RemoveStatus(_Summon, "GUST_OF_WIND_UNWALKABLE");
NOT DB_GustOfWindCurrentSolidWalls(_Source, _Summon);

PROC
PROC_GustOfWindClearWalls()
AND
DB_GustOfWindOldSolidWalls((GUIDSTRING)_Source, (GUIDSTRING)_Summon)
THEN
NOT DB_GustOfWindOldSolidWalls(_Source, _Summon);
