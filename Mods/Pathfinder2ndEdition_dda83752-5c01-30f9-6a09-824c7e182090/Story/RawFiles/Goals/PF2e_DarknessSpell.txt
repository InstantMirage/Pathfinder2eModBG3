//REGION Torches
// Whenever an object tagged as a torch enters the area,
// it loses the burning status, but needs to get it back
// when it leaves the aura.
IF
StatusApplied((GUIDSTRING)_Torch, (STRING)_Status, _, _)
AND
DB_DarknessStatuses(_Status)
AND
DB_TorchBurningTags((TAG)_Tag)
AND
IsTagged(_Torch, _Tag, 1)
AND
HasActiveStatus(_Torch, "BURNING", 1)
THEN
RemoveStatus(_Torch, "BURNING");
DB_DarknessMutedTorches(_Torch);

IF
StatusRemoved((GUIDSTRING)_Torch, (STRING)_Status, _, _)
AND
DB_DarknessStatuses(_Status)
AND
DB_DarknessMutedTorches(_Torch)
THEN
ApplyStatus(_Torch, "BURNING", -1.0);
NOT DB_DarknessMutedTorches(_Torch);

// Torches that become lit in the darkness are unlit
// immediately.
IF
StatusApplied((GUIDSTRING)_Torch, "BURNING", _, _)
AND
DB_TorchBurningTags((TAG)_Tag)
AND
IsTagged(_Torch, _Tag, 1)
AND
DB_DarknessStatuses((STRING)_Status)
AND
HasActiveStatus(_Torch, _Status, 1)
THEN
RemoveStatus(_Torch, "BURNING");

// We also do this process for torches carried in
// inventories.
IF
StatusApplied((GUIDSTRING)_Holder, (STRING)_Status, _, _)
AND
DB_DarknessStatuses(_Status)
THEN
IterateInventory(_Holder, "DarknessInventoryCheck", "Dummy");

IF
EntityEvent((GUIDSTRING)_Torch, "DarknessInventoryCheck")
AND
DB_TorchBurningTags((TAG)_Tag)
AND
IsTagged(_Torch, _Tag, 1)
AND
HasActiveStatus(_Torch, "BURNING", 1)
THEN
RemoveStatus(_Torch, "BURNING");
DB_DarknessMutedTorches(_Torch);

IF
StatusRemoved((GUIDSTRING)_Holder, (STRING)_Status, _, _)
AND
DB_DarknessStatuses(_Status)
THEN
IterateInventory(_Holder, "DarknessRemoveInventoryCheck", "Dummy");

IF
EntityEvent((GUIDSTRING)_Torch, "DarknessRemoveInventoryCheck")
AND
DB_DarknessMutedTorches(_Torch)
THEN
ApplyStatus(_Torch, "BURNING", -1.0);
NOT DB_DarknessMutedTorches(_Torch);

// Torches that become lit in the darkness are unlit
// immediately.
IF
StatusApplied((GUIDSTRING)_Torch, "BURNING", _, _)
AND
DB_TorchBurningTags((TAG)_Tag)
AND
IsTagged(_Torch, _Tag, 1)
AND
GetInventoryOwner(_Torch, (GUIDSTRING)_Owner)
AND
DB_DarknessStatuses((STRING)_Status)
AND
HasActiveStatus(_Owner, _Status, 1)
THEN
RemoveStatus(_Torch, "BURNING");

// Definitely some edge cases when in two separate darkness
// auras of different ranks. Hard to fix, ignoring for now.
//END_REGION
